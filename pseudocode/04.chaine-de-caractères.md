Exo 2estPalindrome(chaine)
Retourne true si la chaîne se lit pareil dans les deux sens.
estPalindrome("kayak")   → true
estPalindrome("bonjour") → false

function estPalindrome(chaine) {

    let chaine = "kayak";

    let inverse = chaine.split("").reverse().join("");

    return inverse === chaine;
}

console.log(estPalindrome("kayak"));
console.log(estPalindrome("Bonjour"));
console.log(estPalindrome("dweedweed"));
